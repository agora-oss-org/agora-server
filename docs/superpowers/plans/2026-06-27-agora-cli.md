# Agora CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `agora`, a single Go binary run on the deployment host: a live terminal dashboard plus host/docker/db/ops command control over a self-hosted Agora Docker Compose deployment.

**Architecture:** One spine, two front-ends. **Providers** (`containerDriver`, `hostProbe`, `apiClient`, `pg`) are the only things that touch the outside world. **Capability modules** (`doctor`, `host`, `db`, `ops`) each register a Cobra command tree and (later) a Bubbletea dashboard panel, fed by the same providers so the CLI and dashboard never drift. Providers are interfaces, so every module is unit-tested against fakes with no Docker/DB/network.

**Tech Stack:** Go 1.25 · [Cobra](https://github.com/spf13/cobra) (commands) · [Bubbletea](https://github.com/charmbracelet/bubbletea) + [Lipgloss](https://github.com/charmbracelet/lipgloss) (TUI) · [gopsutil](https://github.com/shirou/gopsutil) (host probe) · stdlib `net/http` + `net/http/httptest` (api client + tests).

## Global Constraints

- **Go module path:** `github.com/jenova-marie/agora-server/cli` (own `go.mod` under `cli/`).
- **Go version floor:** `go 1.25` (installed: go1.25.0).
- **On-box only.** No SSH, no remote docker context, no `--context` flag. Local `docker.sock`, local host, `localhost` API.
- **Glitter, not load-bearing.** Never reimplement DB logic that a `.mjs` script already owns — shell out to it. The binary must add zero runtime dependency for the deployment.
- **Provider boundary rule.** Only files under `internal/providers/**` may import `os/exec`, `net/http`, `gopsutil`, or touch the filesystem/network. Modules depend on provider **interfaces** only.
- **Security:** never print secret *values*. Report presence/length only. `db` tier commands are confirm-gated (`--yes` to bypass) and require `DATABASE_URL`.
- **API facts (verbatim):** health is `GET /health` → `{ "ok": true, "service": "agora", "version": "v7", "source": "..." }`. Default API port is `env.PORT` (dev default `4000`). API base path is `/v7/:projectId/...`.
- **Compose service names (verbatim):** `agora`, `redis`, `db`, `minio`, `secure-chat`, `scorer-toxicity`, `scorer-relationship`, `scorer-worker`, `neo4j`, `proxy`, `cron`.
- **Commit style:** DCO sign-off (`git commit -s`). End commit messages with `Co-Authored-By: Claude Opus 4.8 (1M context) <noreply@anthropic.com>`.
- **Test command:** `go test ./...` from `cli/`. Build: `go build -o bin/agora .` from `cli/`.

package ui

import (
	"fmt"
	"time"
)

// Bytes renders a byte count in binary units (B, KiB, MiB, GiB, TiB).
func Bytes(n uint64) string {
	const unit = 1024
	if n < unit {
		return fmt.Sprintf("%d B", n)
	}
	units := []string{"KiB", "MiB", "GiB", "TiB", "PiB"}
	val := float64(n)
	i := -1
	for val >= unit && i < len(units)-1 {
		val /= unit
		i++
	}
	return fmt.Sprintf("%.1f %s", val, units[i])
}

// Percent renders used/total as a whole-number percent; total==0 yields "—".
func Percent(used, total uint64) string {
	if total == 0 {
		return "—"
	}
	return fmt.Sprintf("%d%%", used*100/total)
}

// Duration renders a coarse human duration: days+hours, or minutes+seconds, etc.
func Duration(d time.Duration) string {
	if d >= 24*time.Hour {
		days := int(d / (24 * time.Hour))
		hours := int((d % (24 * time.Hour)) / time.Hour)
		return fmt.Sprintf("%dd%dh", days, hours)
	}
	if d >= time.Hour {
		h := int(d / time.Hour)
		m := int((d % time.Hour) / time.Minute)
		return fmt.Sprintf("%dh%dm", h, m)
	}
	if d >= time.Minute {
		m := int(d / time.Minute)
		s := int((d % time.Minute) / time.Second)
		return fmt.Sprintf("%dm%ds", m, s)
	}
	return fmt.Sprintf("%ds", int(d/time.Second))
}
```

package config

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

type Config struct {
	RepoRoot      string
	ComposeFile   string
	DatabaseURL   string
	APIBaseURL    string
	OperatorToken string
}

// ParseDotenv parses minimal KEY=VALUE lines, ignoring blanks and # comments,
// stripping a single layer of surrounding single/double quotes. Empty values omitted.
func ParseDotenv(content string) map[string]string {
	out := map[string]string{}
	for _, line := range strings.Split(content, "\n") {
		line = strings.TrimSpace(line)
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		k, v, ok := strings.Cut(line, "=")
		if !ok {
			continue
		}
		k = strings.TrimSpace(k)
		v = strings.TrimSpace(v)
		v = strings.Trim(v, `"'`)
		if v == "" {
			continue
		}
		out[k] = v
	}
	return out
}

// Load resolves config by walking up from startDir to find docker-compose.yml,
// reading the sibling .env, and overlaying real environment (env wins).
func Load(startDir string, env func(string) string) (Config, error) {
	root, compose, err := findCompose(startDir)
	if err != nil {
		return Config{}, err
	}
	dotenv := map[string]string{}
	if b, err := os.ReadFile(filepath.Join(root, ".env")); err == nil {
		dotenv = ParseDotenv(string(b))
	}
	get := func(k string) string {
		if v := env(k); v != "" {
			return v
		}
		return dotenv[k]
	}
	port := get("PORT")
	if port == "" {
		port = "4000"
	}
	return Config{
		RepoRoot:      root,
		ComposeFile:   compose,
		DatabaseURL:   get("DATABASE_URL"),
		APIBaseURL:    fmt.Sprintf("http://localhost:%s", port),
		OperatorToken: get("AGORA_OPERATOR_TOKEN"),
	}, nil
}

func findCompose(start string) (root, compose string, err error) {
	dir := start
	for {
		candidate := filepath.Join(dir, "docker-compose.yml")
		if _, statErr := os.Stat(candidate); statErr == nil {
			return dir, candidate, nil
		}
		parent := filepath.Dir(dir)
		if parent == dir {
			return "", "", errors.New("docker-compose.yml not found in this or any parent directory")
		}
		dir = parent
	}
}
```

package host

import (
	"context"
	"time"
)

type Snapshot struct {
	MemUsed, MemTotal   uint64
	DiskUsed, DiskTotal uint64
	CPUPercent          float64
	Uptime              time.Duration
}

// DiskHeadroomOK reports whether free disk is at least minFreePct percent.
func (s Snapshot) DiskHeadroomOK(minFreePct int) bool {
	if s.DiskTotal == 0 {
		return false
	}
	free := s.DiskTotal - s.DiskUsed
	return free*100/s.DiskTotal >= uint64(minFreePct)
}

type HostProbe interface {
	Snapshot(ctx context.Context) (Snapshot, error)
}

// Fake is a deterministic HostProbe for tests.
type Fake struct {
	Snap Snapshot
	Err  error
}

func (f Fake) Snapshot(context.Context) (Snapshot, error) { return f.Snap, f.Err }
```

package host

import (
	"context"
	"time"

	"github.com/shirou/gopsutil/v4/cpu"
	"github.com/shirou/gopsutil/v4/disk"
	gohost "github.com/shirou/gopsutil/v4/host"
	"github.com/shirou/gopsutil/v4/mem"
)

type gopsutilProbe struct{}

// New returns a HostProbe backed by gopsutil reading this machine.
func New() HostProbe { return gopsutilProbe{} }

func (gopsutilProbe) Snapshot(ctx context.Context) (Snapshot, error) {
	var s Snapshot
	if vm, err := mem.VirtualMemoryWithContext(ctx); err == nil {
		s.MemUsed, s.MemTotal = vm.Used, vm.Total
	}
	if du, err := disk.UsageWithContext(ctx, "/"); err == nil {
		s.DiskUsed, s.DiskTotal = du.Used, du.Total
	}
	if pcts, err := cpu.PercentWithContext(ctx, 0, false); err == nil && len(pcts) > 0 {
		s.CPUPercent = pcts[0]
	}
	if up, err := gohost.UptimeWithContext(ctx); err == nil {
		s.Uptime = time.Duration(up) * time.Second
	}
	return s, nil
}
```

import (
	"context"
	"encoding/json"
	"io"
	"strings"
)

type Service struct {
	Name   string `json:"Name"`
	State  string `json:"State"`
	Health string `json:"Health"`
}

type ContainerDriver interface {
	PS(ctx context.Context) ([]Service, error)
	Logs(ctx context.Context, svc string, follow bool, tail int) (io.ReadCloser, error)
	Up(ctx context.Context, profiles []string) error
	Down(ctx context.Context) error
	Restart(ctx context.Context, svc string) error
}

// parsePS parses the line-delimited JSON of `docker compose ps --format json`.
func parsePS(b []byte) ([]Service, error) {
	var out []Service
	for _, line := range strings.Split(strings.TrimSpace(string(b)), "\n") {
		line = strings.TrimSpace(line)
		if line == "" {
			continue
		}
		var s Service
		if err := json.Unmarshal([]byte(line), &s); err != nil {
			return nil, err
		}
		out = append(out, s)
	}
	return out, nil
}

// Fake records lifecycle calls for module tests.
type Fake struct {
	Services []Service
	Calls    []string
	Err      error
}

func (f *Fake) PS(context.Context) ([]Service, error) { return f.Services, f.Err }
func (f *Fake) Logs(context.Context, string, bool, int) (io.ReadCloser, error) {
	return io.NopCloser(strings.NewReader("")), f.Err
}
func (f *Fake) Up(_ context.Context, profiles []string) error {
	f.Calls = append(f.Calls, "up:"+strings.Join(profiles, ","))

}

package container

import (
	"context"
	"io"
	"os/exec"
	"strconv"
)

type compose struct{ file string }

// NewCompose returns a ContainerDriver that shells out to `docker compose -f <file>`.
func NewCompose(composeFile string) ContainerDriver { return compose{file: composeFile} }

func (c compose) base(args ...string) []string {
	return append([]string{"compose", "-f", c.file}, args...)
}

func (c compose) PS(ctx context.Context) ([]Service, error) {
	out, err := exec.CommandContext(ctx, "docker", c.base("ps", "--format", "json")...).Output()
	if err != nil {
		return nil, err
	}
	return parsePS(out)
}

func (c compose) Logs(ctx context.Context, svc string, follow bool, tail int) (io.ReadCloser, error) {
	args := c.base("logs", "--tail", strconv.Itoa(tail))
	if follow {
		args = append(args, "--follow")
	}
	args = append(args, svc)
	cmd := exec.CommandContext(ctx, "docker", args...)
	pr, pw := io.Pipe()
	cmd.Stdout = pw
	cmd.Stderr = pw
	if err := cmd.Start(); err != nil {
		return nil, err
	}
	go func() { pw.CloseWithError(cmd.Wait()) }()
	return pr, nil
}

func (c compose) Up(ctx context.Context, profiles []string) error {
	args := []string{"compose", "-f", c.file}
	for _, p := range profiles {
		args = append(args, "--profile", p)
	}
	args = append(args, "up", "-d")
	return exec.CommandContext(ctx, "docker", args...).Run()
}

func (c compose) Down(ctx context.Context) error {
	return exec.CommandContext(ctx, "docker", c.base("down")...).Run()
}

func (c compose) Restart(ctx context.Context, svc string) error {
	return exec.CommandContext(ctx, "docker", c.base("restart", svc)...).Run()
}

	token string
	hc    *http.Client

func NewClient(baseURL, token string) *Client {
	return &Client{base: baseURL, token: token, hc: &http.Client{Timeout: 10 * time.Second}}
}

func (c *Client) do(ctx context.Context, method, path string, out any) error {
	req, err := http.NewRequestWithContext(ctx, method, c.base+path, nil)
	if err != nil {
		return err
	}
	if c.token != "" {
		req.Header.Set("Authorization", "Bearer "+c.token)
	}
	resp, err := c.hc.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode >= 400 {
		return fmt.Errorf("api %s %s: status %d", method, path, resp.StatusCode)
	}
	if out == nil {
		return nil
	}
	return json.NewDecoder(resp.Body).Decode(out)

package doctor

import "context"

type Status int

const (
	Pass Status = iota
	Warn
	Fail
)

type Result struct {
	Name   string
	Status Status
	Detail string
}

type Check struct {
	Name string
	Run  func(context.Context) Result
}

// Run executes each check in order and collects results.
func Run(ctx context.Context, checks []Check) []Result {
	out := make([]Result, 0, len(checks))
	for _, c := range checks {
		out = append(out, c.Run(ctx))
	}
	return out
}

package doctor

import (
	"fmt"
	"strings"

	"github.com/jenova-marie/agora-server/cli/internal/ui"

func glyph(s Status) string {
	switch s {
	case Pass:
		return ui.OK.Render("✔")
	case Warn:
		return ui.Warn.Render("●")
	default:
		return ui.Fail.Render("✘")
	}
}

// Report renders results as a pretty aligned list.
func Report(results []Result) string {
	var b strings.Builder
	for _, r := range results {
		line := fmt.Sprintf("%s  %s", glyph(r.Status), r.Name)
		if r.Detail != "" {
			line += ui.Muted.Render("  — " + r.Detail)
		}
		b.WriteString(line + "\n")
	}
	return b.String()
}

			if _, err := exec.LookPath("docker"); err != nil {

			}

					return Result{Name: "containers", Status: Warn, Detail: s.Name + " " + s.State}

type runner func(name string, args ...string) error

// confirm reads one line and returns true only on y/yes (fail closed).
func confirm(in io.Reader, prompt string) bool {
	fmt.Print(prompt + " [y/N] ")
	line, _ := bufio.NewReader(in).ReadString('\n')
	line = strings.ToLower(strings.TrimSpace(line))
	return line == "y" || line == "yes"
}

// scriptCmd builds a fenced cobra command that shells out to a maintenance .mjs.
func scriptCmd(repoRoot, use, script string, run runner) *cobra.Command {
	var yes bool
	cmd := &cobra.Command{
		Use:   use,
		Short: "Run " + script + " (bypasses the server — fenced)",
		RunE: func(cmd *cobra.Command, args []string) error {
			if !yes && !confirm(cmd.InOrStdin(), fmt.Sprintf("Run %s against the local DB?", script)) {
				return fmt.Errorf("aborted")
			}
			name, scriptArgs := pg.ScriptCmd(repoRoot, script)
			return run(name, scriptArgs...)
		},
	}
	cmd.Flags().BoolVar(&yes, "yes", false, "skip confirmation")
	return cmd
}

	run := func(name string, args ...string) error { return execRun(name, args...) }

package dashboard

import (
	"context"
	"time"

	tea "github.com/charmbracelet/bubbletea"
	"github.com/jenova-marie/agora-server/cli/internal/providers/container"
	"github.com/jenova-marie/agora-server/cli/internal/providers/host"
)

type tickMsg struct{}

type dataMsg struct {
	services []container.Service
	snap     host.Snapshot
	err      error
}

type model struct {
	services []container.Service
	snap     host.Snapshot
	err      error
	drv      container.ContainerDriver
	probe    host.HostProbe
}

func (m model) Init() tea.Cmd { return tea.Batch(m.refresh(), tick()) }

func tick() tea.Cmd {
	return tea.Tick(2*time.Second, func(time.Time) tea.Msg { return tickMsg{} })
}

func (m model) refresh() tea.Cmd {
	return func() tea.Msg {
		ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
		defer cancel()
		svcs, err := m.drv.PS(ctx)
		snap, _ := m.probe.Snapshot(ctx)
		return dataMsg{services: svcs, snap: snap, err: err}
	}
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case dataMsg:
		m.services, m.snap, m.err = msg.services, msg.snap, msg.err
		return m, nil
	case tickMsg:
		return m, tea.Batch(m.refresh(), tick())
	case tea.KeyMsg:
		switch msg.String() {
		case "q", "ctrl+c":
			return m, tea.Quit
		}
	}
	return m, nil
}
